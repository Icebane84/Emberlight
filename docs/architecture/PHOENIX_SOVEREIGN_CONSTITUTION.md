```javascript
function _oslmTopologicalSort(modules) {
    const inDegree = new Map(), adjacency = new Map();
    for (const [id] of modules) { inDegree.set(id, 0); adjacency.set(id, []); }
    for (const [id, mod] of modules)
        for (const dep of mod.requires) {
            adjacency.get(dep)?.push(id);
            inDegree.set(id, (inDegree.get(id) ?? 0) + 1);
        }
    const queue = [], sorted = [];
    for (const [id, degree] of inDegree) if (degree === 0) queue.push(id);
    while (queue.length > 0) {
        const id = queue.shift();
        sorted.push(modules.get(id));
        for (const neighbor of adjacency.get(id) ?? []) {
            const d = inDegree.get(neighbor) - 1;
            inDegree.set(neighbor, d);
            if (d === 0) queue.push(neighbor);
        }
    }
    if (sorted.length !== modules.size)
        throw new Error('ERR_0x03: CIRCULAR_DEPENDENCY -- OSLM DAG contains a cycle');
    return sorted;
}
```

### [SEC-01.4] Hopcroft-Tarjan Cut-Vertex Detection

A **cut-vertex** is any module whose removal disconnects the dependency graph -- a Single Point of Failure (SPOF). The algorithm is **multigraph-hardened** via unique edge ID tokens (`src->dst`) to prevent false positives.

- `WARN_STR_OVERCOMPLEX` -- cut-vertex has valid `provides` contract; boot continues with warning.
- `ERR_STR_INCONSISTENCY` -- cut-vertex has no `provides` contract; boot halted.

```javascript
function _oslmFindCutVertices(modules) {
    const visited = new Map(), cutSet = new Set(), warnings = [];
    let timer = 0;
    function _dfs(nodeId, parentId, edgeId) {
        const disc = timer++;
        visited.set(nodeId, { disc, low: disc });
        let childCount = 0;
        const mod = modules.get(nodeId);
        for (const neighborId of (mod?.requires ?? [])) {
            const uniqueEdge = `${nodeId}\u2192${neighborId}`;
            if (!visited.has(neighborId)) {
                childCount++;
                _dfs(neighborId, nodeId, uniqueEdge);
                const nodeLow = visited.get(nodeId).low;
                const neighborLow = visited.get(neighborId).low;
                visited.get(nodeId).low = Math.min(nodeLow, neighborLow);
                const isRoot = parentId === null;
                if (isRoot && childCount > 1) cutSet.add(nodeId);
                else if (!isRoot && visited.get(neighborId).disc >= visited.get(nodeId).disc) cutSet.add(nodeId);
            } else if (uniqueEdge !== edgeId) {
                visited.get(nodeId).low = Math.min(visited.get(nodeId).low, visited.get(neighborId).disc);
            }
        }
    }
    for (const [id] of modules) if (!visited.has(id)) _dfs(id, null, '');
    for (const id of cutSet) {
        const mod = modules.get(id);
        if (!mod?.provides?.length) warnings.push(`ERR_STR_INCONSISTENCY: Cut-vertex '${id}' has no provides contract`);
        else warnings.push(`WARN_STR_OVERCOMPLEX: Cut-vertex '${id}' -- consider decomposition`);
    }
    return { cutVertices: [...cutSet], warnings };
}
```

### [SEC-05.2] Header Packing

```javascript
/**
 * Packs the 96-byte tripartite header into the binary heap.
 * @param {DataView}   view       - DataView over the 2,048-byte ArrayBuffer
 * @param {bigint}     tickCount  - Monotonic tick count
 * @param {number}     prngSeed   - Mulberry32 PRNG seed state
 * @param {ByteOffset} payloadLen - Byte length of entity data
 * @param {number}     crc32Val   - CRC32 of entity data
 * @param {Uint8Array} cartUUID   - 12-byte cartridge UUID
 * @param {number}     epoch      - Unix epoch seconds
 * @param {number}     capMask    - Granted capability bitmask
 * @param {number}     authTier   - Authority tier (1-4)
 * @param {Uint8Array} sgiHash    - 16-byte SGI feature hash
 * @param {number}     auditSalt  - Mutation nonce
 * @returns {void}
 */
function _packHeader(view, tickCount, prngSeed, payloadLen, crc32Val,
                     cartUUID, epoch, capMask, authTier, sgiHash, auditSalt) {
    view.setUint32(0x000, HEAP_MAGIC,   true);
    view.setBigInt64(0x004, tickCount,  true);
    view.setUint32(0x00C, prngSeed,     true);
    view.setUint32(0x010, payloadLen,   true);
    view.setUint32(0x014, crc32Val,     true);
    view.setUint32(0x018, 0x00000001,   true); // format version
    view.setUint32(0x01C, 0,            true); // reserved
    const u8 = new Uint8Array(view.buffer);
    u8.set(cartUUID.subarray(0, 12), 0x020);
    view.setUint32(0x02C, epoch,        true);
    view.setUint32(0x040, capMask,      true);
    view.setUint8(0x044, authTier);
    u8.fill(0, 0x045, 0x048);
    u8.set(sgiHash.subarray(0, 16),    0x048);
    view.setUint32(0x058, auditSalt,    true);
    u8.fill(0, 0x05C, 0x060);
}
```

### [SEC-05.5] Reverse-Topological Merkle-DAG Compaction

Content-addressed SHA-256 hashes evaluated in **reverse topological order** (leaves first, root last) so a change in a leaf automatically invalidates all ancestors without a full heap scan.

```javascript
/**
 * @param {Map<string, { data: Uint8Array, children: string[] }>} assetGraph
 * @param {string[]} topoOrder - Topological order (root first)
 * @returns {Promise<Map<string, ArrayBuffer>>} nodeId to SHA-256 hash
 */
async function _computeMerkleDAG(assetGraph, topoOrder) {
    const hashes = new Map();
    for (const nodeId of [...topoOrder].reverse()) { // leaves first
        const node = assetGraph.get(nodeId);
        if (!node) continue;
        const childBufs = node.children.map(c => hashes.get(c) ?? new ArrayBuffer(0));
        const totalLen  = node.data.byteLength + childBufs.reduce((s, b) => s + b.byteLength, 0);
        const combined  = new Uint8Array(totalLen);
        combined.set(node.data, 0);
        let offset = node.data.byteLength;
        for (const b of childBufs) { combined.set(new Uint8Array(b), offset); offset += b.byteLength; }
        hashes.set(nodeId, await crypto.subtle.digest('SHA-256', combined));
    }
    return hashes;
}
```

## [SEC-08] Viewport Engine, Rendering & Load Shedder

### [SEC-08.1] Deterministic 3-Stage Viewport Pipeline

```plaintext
STAGE 1: SPATIAL DEADZONE
  If |target - camera| < DEADZONE (100x80 units), camera does not move.

STAGE 2: VERTICAL HYSTERESIS & ASYMMETRIC DAMPING
  Upward movement: higher damping factor. Downward: lower. Prevents vertical snap.

STAGE 3: 2ND-ORDER KINETIC LOOK-AHEAD
  P_lookahead = P_current + v*dt + 0.5*a*dt^2
```

```javascript
const DEADZONE_W = 100, DEADZONE_H = 80;
/**
 * @param {{ x:number, y:number }} camPos
 * @param {{ x:number, y:number }} camVel
 * @param {{ x:number, y:number }} camAcc
 * @param {{ x:number, y:number }} targetPos
 * @param {Seconds} dt
 * @returns {{ pos:{x:number,y:number}, vel:{x:number,y:number} }}
 */
function _viewportTick(camPos, camVel, camAcc, targetPos, dt) {
    const dx = targetPos.x - camPos.x, dy = targetPos.y - camPos.y;
    const inDeadzone   = Math.abs(dx) < DEADZONE_W/2 && Math.abs(dy) < DEADZONE_H/2;
    const effTargetX   = inDeadzone ? camPos.x : targetPos.x;
    const effTargetY   = inDeadzone ? camPos.y : targetPos.y;
    const dampY        = (effTargetY > camPos.y) ? 0.85 : 0.95; // asymmetric
    const dampX        = 0.90;
    const newVelX      = camVel.x * dampX + (effTargetX - camPos.x) * (1 - dampX);
    const newVelY      = camVel.y * dampY + (effTargetY - camPos.y) * (1 - dampY);
    const newPosX      = camPos.x + newVelX * dt + 0.5 * camAcc.x * dt * dt;
    const newPosY      = camPos.y + newVelY * dt + 0.5 * camAcc.y * dt * dt;
    return { pos: { x: newPosX, y: newPosY }, vel: { x: newVelX, y: newVelY } };
}
```

```javascript
/**
 * @typedef {Object} HunkDescriptor
 * @property {number}  startLine     - 1-indexed start of target region
 * @property {number}  endLine       - 1-indexed end of target region (inclusive)
 * @property {string}  targetContent - Exact string to replace (must be unique in range)
 * @property {string}  replacement   - Replacement content
 * @property {boolean} allowMultiple - If false, fails if targetContent appears >1 time
 */

/**
 * @param {string}         source
 * @param {HunkDescriptor} hunk
 * @returns {{ ok: boolean, result: string, error?: string }}
 */
function applyHunk(source, hunk) {
    const lines  = source.split('\n');
    const region = lines.slice(hunk.startLine - 1, hunk.endLine).join('\n');
    const occurs = region.split(hunk.targetContent).length - 1;
    if (occurs === 0)                         return { ok: false, result: source, error: 'targetContent not found' };
    if (occurs > 1 && !hunk.allowMultiple)    return { ok: false, result: source, error: 'targetContent not unique' };
    const patched = region.replace(hunk.targetContent, hunk.replacement);
    const result  = [...lines.slice(0, hunk.startLine - 1), patched, ...lines.slice(hunk.endLine)].join('\n');
    return { ok: true, result };
}
/** @param {HunkDescriptor} hunk @param {string} reason @returns {void} */
function rejectHunk(hunk, reason) { _logTelemetry(`HUNK REJECTED [L${hunk.startLine}-L${hunk.endLine}]: ${reason}`); }
```

### [SEC-13.6] `synarche_parser.js` Canonical API & Reference Specification

The STCP toolchain is embodied in `synarche_parser.js`, which exposes four pure, deterministic primitives:

```javascript
/**
 * @typedef {Object} Token
 * @property {'KEYWORD'|'IDENTIFIER'|'LITERAL'|'SYMBOL'|'DIRECTIVE'|'WHITESPACE'|'COMMENT'} type
 * @property {string} value
 * @property {number} start - 0-indexed byte offset
 * @property {number} end   - 0-indexed byte offset
 * @property {number} line  - 1-indexed line number
 * @property {number} col   - 1-indexed column number
 */

/**
 * @typedef {Object} CompileResult
 * @property {boolean} ok
 * @property {string} [code]               - Emitted VSRP-001 JavaScript source
 * @property {Object} [ast]                - Validated AST
 * @property {string[]} [capabilities]     - Extracted capability list
 * @property {string[]} [errors]           - Diagnostic errors with line/column
 * @property {number} [memoryBytesRequired]- Exact bytes needed in PERSIST-001 partition
 */

const SynarcheCompiler = {
    /**
     * Plane 1: Tokenize source stream
     * @param {string} source
     * @param {{ preserveTrivia?: boolean }} [options]
     * @returns {Token[]}
     */
    tokenize(source, options) { /* pure token scanner */ },

    /**
     * Plane 2: Parse tokens into validated AST
     * @param {Token[]} tokens
     * @param {{ mode: 'STRICT'|'SLOPPY' }} options
     * @returns {{ ast: Object, errors: string[] }}
     */
    parse(tokens, options) { /* AST & mixin linearizer */ },

    /**
     * Plane 3: Transpile AST to VSRP-001 Cartridge IIFE
     * @param {Object} ast
     * @param {{ format: 'IIFE'|'CJS' }} [options]
     * @returns {string}
     */
    emit(ast, options) { /* GUCA code emitter */ },

    /**
     * Complete Tri-Planar End-to-End Pipeline
     * @param {string} source
     * @param {{ mode?: 'STRICT'|'SLOPPY', format?: 'IIFE'|'CJS' }} [options]
     * @returns {CompileResult}
     */
    compile(source, options) {
        const tokens = this.tokenize(source);
        const { ast, errors } = this.parse(tokens, { mode: options?.mode || 'STRICT' });
        if (errors.length > 0 && options?.mode !== 'SLOPPY') {
            return { ok: false, errors };
        }
        const code = this.emit(ast, { format: options?.format || 'IIFE' });
        return {
            ok: true,
            code,
            ast,
            capabilities: ast.capabilities || [],
            memoryBytesRequired: ast.memoryLayout?.totalBytes || 0
        };
    }
};
```
